# Map Tab 3D GLB Model Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the MapLibre Map tab with a native RealityKit viewer showing the maps3d.io 3D model of the event area, with tappable POI markers and a live user-location dot.

**Architecture:** One-time offline GLB→USDZ conversion (Blender headless); USDZ bundled as an app resource. `Map3DScreen` hosts a SwiftUI `RealityView` with built-in orbit camera controls. Pure geo math (`Map3DGeo`) linearly maps lat/lng (via EPSG:3857) onto the model's measured bounding box; markers and the GPS dot are RealityKit entities placed with that mapping, with raycast-to-terrain for height. MapLibre is removed at the end.

**Tech Stack:** SwiftUI, RealityKit (`RealityView`, iOS 18), CoreLocation, Blender 5.1 CLI (asset step only), xcodegen.

Create `WBW/Map3DGeo.swift`:

```swift
import Foundation
import simd

/// แปลงพิกัดโลก lat/lng → พิกัด local บนโมเดล maps3d (Tab แผนที่ 3D)
/// กรอบจาก docs/map3d/maps3d-settings.json · projection EPSG:3857
enum Map3DGeo {
    // Bounding box ของพื้นที่โมเดล (hexagon bbox)
    static let swLat = 20.02371, swLng = 99.88272
    static let neLat = 20.06727, neLng = 99.92288

    private static let earthR = 6378137.0

    /// web-mercator EPSG:3857 (หน่วยเมตร)
    static func mercator(lat: Double, lng: Double) -> SIMD2<Double> {
        let x = earthR * lng * .pi / 180
        let y = earthR * log(tan(.pi / 4 + lat * .pi / 360))
        return SIMD2(x, y)
    }

    /// กรอบโมเดลจริง — วัดจาก visualBounds ตอนโหลด USDZ
    struct ModelFrame: Equatable {
        var minX: Float, maxX: Float, minZ: Float, maxZ: Float
        /// RealityKit แกน Z ชี้เข้าหากล้อง: ทิศเหนือของโมเดล = -Z (ค่า default; ยืนยันด้วยตาใน Task 4)
        var flipZ: Bool = true

        init(minX: Float, maxX: Float, minZ: Float, maxZ: Float, flipZ: Bool = true) {
            self.minX = minX; self.maxX = maxX; self.minZ = minZ; self.maxZ = maxZ; self.flipZ = flipZ
        }
    }

    /// พิกัดโลก → (x, z) บนโมเดล · nil = อยู่นอกกรอบพื้นที่
    static func modelXZ(lat: Double, lng: Double, frame: ModelFrame) -> SIMD2<Float>? {
        guard lat >= swLat, lat <= neLat, lng >= swLng, lng <= neLng else { return nil }
        let sw = mercator(lat: swLat, lng: swLng)
        let ne = mercator(lat: neLat, lng: neLng)
        let p = mercator(lat: lat, lng: lng)
        let u = Float((p.x - sw.x) / (ne.x - sw.x))   // 0 = ตะวันตก, 1 = ตะวันออก
        var v = Float((p.y - sw.y) / (ne.y - sw.y))   // 0 = ใต้, 1 = เหนือ
        if frame.flipZ { v = 1 - v }                   // เหนือ → z ต่ำ (-Z)
        return SIMD2(frame.minX + u * (frame.maxX - frame.minX),
                     frame.minZ + v * (frame.maxZ - frame.minZ))
    }
}
```

```swift
import SwiftUI
import RealityKit

/// แผนที่ 3D — โมเดล USDZ จาก maps3d.io แทน MapLibre เดิม
/// หมุน/ซูมด้วย orbit controls ของ RealityKit (iOS 18)
struct Map3DScreen: View {
    @State private var loadFailed = false
    @State private var retryCount = 0

    var body: some View {
        ZStack(alignment: .bottomLeading) {
            if loadFailed {
                VStack(spacing: 12) {
                    Image(systemName: "map.circle").font(.system(size: 44)).foregroundStyle(.secondary)
                    Text("โหลดแผนที่ 3D ไม่สำเร็จ")
                    Button("ลองใหม่") { loadFailed = false; retryCount += 1 }
                        .buttonStyle(.borderedProminent).tint(Color.wbwGold)
                }
            } else {
                Map3DViewer(loadFailed: $loadFailed)
                    .id(retryCount)          // สร้าง RealityView ใหม่ตอนกดลองใหม่
                    .ignoresSafeArea()
            }

            // เงื่อนไข license ของ maps3d — ต้องแสดง attribution เสมอ
            Text("แผนที่ 3D: Satlas · Allen Institute for AI · © OpenStreetMap contributors")
                .font(.system(size: 9))
                .foregroundStyle(.white.opacity(0.85))
                .shadow(radius: 1)
                .padding(.leading, 10)
                .padding(.bottom, 4)
        }
    }
}

private struct Map3DViewer: View {
    @Binding var loadFailed: Bool

    var body: some View {
        RealityView { content in
            do {
                guard let url = Bundle.main.url(forResource: "map3d", withExtension: "usdz") else {
                    throw URLError(.fileDoesNotExist)
                }
                let map = try await Entity(contentsOf: url)
                content.add(map)

                let bounds = map.visualBounds(relativeTo: nil)

                // กล้องเริ่มต้น: เฉียง ~55° มองกลางโมเดล (เลียนมุม MapLibre เดิม)
                let cam = PerspectiveCamera()
                let size = max(bounds.extents.x, bounds.extents.z)
                let dist = size * 1.1
                let pitch: Float = 55 * .pi / 180
                let camPos = SIMD3<Float>(bounds.center.x,
                                          bounds.center.y + dist * sin(pitch),
                                          bounds.center.z + dist * cos(pitch))
                cam.look(at: bounds.center, from: camPos, relativeTo: nil)
                content.add(cam)

                // แสงหลัก — USDZ เป็น PBR ต้องมีแสงเอง
                let sun = DirectionalLight()
                sun.light.intensity = 5000
                sun.orientation = simd_quatf(angle: -.pi / 3, axis: SIMD3<Float>(1, 0, 0))
                content.add(sun)

                // MARK: markers ต่อใน Task 4
            } catch {
                loadFailed = true
            }
        }
        .realityViewCameraControls(.orbit)
    }
}
```

- [ ] **Step 1: Write Map3DMarkers.swift**

```swift
import Foundation
import RealityKit
import UIKit

/// จุดสำคัญในงาน — hardcode ในแอป (ไม่ดึงจาก backend)
struct Map3DPOI: Identifiable, Equatable {
    let name: String
    let lat: Double
    let lng: Double
    var id: String { name }

    // ⚠️ พิกัด placeholder อยู่ในกรอบโมเดล — รอพิกัดจริงจากทีมงานแล้วแก้ตรงนี้ที่เดียว
    static let all: [Map3DPOI] = [
        Map3DPOI(name: "เวทีกลาง", lat: 20.0455, lng: 99.9028),
        Map3DPOI(name: "จุดเช็คอิน", lat: 20.0500, lng: 99.8950),
        Map3DPOI(name: "จุดปฐมพยาบาล", lat: 20.0400, lng: 99.9100),
    ]
}

enum Map3DMarkerBuilder {
    static let markerPrefix = "poi:"

    /// สร้างหมุด + ป้ายชื่อ (billboard) ต่อจุด แล้วแขวนใต้ root ของโมเดล
    /// scale = ขนาดสัมพัทธ์กับโมเดล (คำนวณจาก visualBounds ฝั่งผู้เรียก)
    static func attach(to map: Entity, pois: [Map3DPOI], frame: Map3DGeo.ModelFrame, scale: Float) {
        let topY = map.visualBounds(relativeTo: nil).max.y
        for poi in pois {
            guard let xz = Map3DGeo.modelXZ(lat: poi.lat, lng: poi.lng, frame: frame) else { continue }
            let root = Entity()
            root.name = markerPrefix + poi.name

            let pin = ModelEntity(mesh: .generateSphere(radius: 0.9 * scale),
                                  materials: [UnlitMaterial(color: UIColor.systemRed)])
            pin.position.y = 0.9 * scale
            root.addChild(pin)

            let label = ModelEntity(
                mesh: .generateText(poi.name, extrusionDepth: 0.02 * scale,
                                    font: .systemFont(ofSize: CGFloat(1.6 * scale), weight: .semibold),
                                    alignment: .center),
                materials: [UnlitMaterial(color: .white)])
            // จัดกึ่งกลางป้าย (generateText วาง origin ที่ baseline ซ้าย)
            label.position = SIMD3<Float>(-label.visualBounds(relativeTo: nil).extents.x / 2,
                                          2.6 * scale, 0)
            let labelPivot = Entity()          // pivot กลางให้ billboard หมุนสวย
            labelPivot.addChild(label)
            labelPivot.components.set(BillboardComponent())
            root.addChild(labelPivot)

            root.components.set(InputTargetComponent())
            root.components.set(CollisionComponent(shapes: [.generateSphere(radius: 2.5 * scale)]))

            // ความสูง: raycast หาผิวโมเดลใน Task step ถัดไป — ระหว่างนี้วางที่ยอดโมเดล
            root.position = SIMD3<Float>(xz.x, surfaceY(x: xz.x, z: xz.y, map: map) ?? topY, xz.y)
            map.addChild(root)
        }
    }

    /// raycast ดิ่งลงหาผิวโมเดล — ต้องมี CollisionComponent บน mesh (ดู addStaticCollision)
    static func surfaceY(x: Float, z: Float, map: Entity) -> Float? {
        guard let scene = map.scene else { return nil }
        let top = map.visualBounds(relativeTo: nil).max.y
        let hits = scene.raycast(origin: SIMD3<Float>(x, top + 10, z),
                                 direction: SIMD3<Float>(0, -1, 0),
                                 length: (top + 20) * 2, query: .nearest,
                                 mask: .all, relativeTo: nil)
        return hits.first?.position.y
    }

    /// gen collision mesh แบบ static ให้ทุก ModelEntity ในโมเดล (ใช้กับ raycast)
    static func addStaticCollision(to entity: Entity) async {
        for child in entity.children { await addStaticCollision(to: child) }
        guard let model = entity as? ModelEntity, let mesh = model.model?.mesh else { return }
        if let shape = try? await ShapeResource.generateStaticMesh(from: mesh) {
            model.components.set(CollisionComponent(shapes: [shape], mode: .default))
        }
    }

    /// entity ที่โดน tap (อาจเป็นลูก) → ชื่อ POI · ไม่ใช่ marker = nil
    static func poiName(for entity: Entity) -> String? {
        var e: Entity? = entity
        while let cur = e {
            if cur.name.hasPrefix(markerPrefix) { return String(cur.name.dropFirst(markerPrefix.count)) }
            e = cur.parent
        }
        return nil
    }
```

- [ ] **Step 1: Write Map3DLocation.swift**

```swift
import Foundation
import CoreLocation

/// ตำแหน่ง GPS ผู้ใช้สำหรับจุดสีฟ้าบนแผนที่ 3D
/// permission มีใน Info.plist แล้ว (NSLocationWhenInUseUsageDescription)
/// โดนปฏิเสธ = coordinate เป็น nil ตลอด — หน้าจอทำงานปกติ ไม่มีจุด
@MainActor
final class Map3DLocation: NSObject, ObservableObject, CLLocationManagerDelegate {
    @Published var coordinate: CLLocationCoordinate2D?
    private let manager = CLLocationManager()

    override init() {
        super.init()
        manager.delegate = self
        manager.desiredAccuracy = kCLLocationAccuracyNearestTenMeters
    }

    func start() {
        manager.requestWhenInUseAuthorization()
        manager.startUpdatingLocation()
    }

    func stop() { manager.stopUpdatingLocation() }

    nonisolated func locationManager(_ manager: CLLocationManager, didUpdateLocations locations: [CLLocation]) {
        let c = locations.last?.coordinate
        Task { @MainActor in self.coordinate = c }
    }

    nonisolated func locationManager(_ manager: CLLocationManager, didFailWithError error: Error) {
        // เงียบไว้ — จุดหายเฉย ๆ ตาม spec
    }
}
```

- [ ] **Step 2: Wire the dot into Map3DScreen**

In `WBW/Map3DScreen.swift`:

1. `Map3DScreen` gains `@StateObject private var location = Map3DLocation()`; pass into `Map3DViewer` as `@ObservedObject var location: Map3DLocation`. Call `.onAppear { location.start() }` / `.onDisappear { location.stop() }` on the viewer.
2. In the `make` closure (after marker attach), store references via two new `@State` vars on `Map3DViewer`... — RealityView `make` cannot write @State directly; instead keep the dot inside the scene graph and find it by name in `update:`. Concretely, at the end of `make`:

```swift

                      materials: [UnlitMaterial(color: UIColor.systemBlue)])
let rim = ModelEntity(mesh: .generateSphere(radius: 1.45 * markerScale),
                      materials: [UnlitMaterial(color: UIColor.white.withAlphaComponent(0.5))])
dot.addChild(rim)
dot.name = "userDot"
dot.isEnabled = false
map.addChild(dot)
map.name = "map3dRoot"
```

3. Give `RealityView` an `update:` closure (second trailing closure of the same `RealityView { ... } update: { ... }` call):

```swift
update: { content in
    guard let map = content.entities.first(where: { $0.name == "map3dRoot" })
            ?? content.entities.first?.findEntity(named: "map3dRoot"),
          let dot = map.findEntity(named: "userDot") else { return }
    let b = map.visualBounds(relativeTo: nil)
    let f = Map3DGeo.ModelFrame(minX: b.min.x, maxX: b.max.x, minZ: b.min.z, maxZ: b.max.z)
    guard let c = location.coordinate,
          let xz = Map3DGeo.modelXZ(lat: c.latitude, lng: c.longitude, frame: f) else {
        dot.isEnabled = false
        return
    }
    let y = Map3DMarkerBuilder.surfaceY(x: xz.x, z: xz.y, map: map) ?? b.max.y
    dot.position = SIMD3<Float>(xz.x, y + 1.2 * (max(b.extents.x, b.extents.z) / 200), xz.y)
    dot.isEnabled = true
}
```
